            width: '10%',
            children: [
                {
                    title: `${calendar.day_th}`,

                    }
                }
            ]
        }

                return ++key
            },
            width: '10%',
        },
        {
            title: 'พนักงาน',
            dataIndex: 'name',
            key: 'name',
            width: '20%',
        },
        ...mapColumnDay
    ];

      return (
        <Table dataSource={dataSource} columns={columns} />
